% LaTeX additions for the PDF version, loaded via include-in-header in _quarto.yml.

% Wrap long code lines instead of running off the page.
\usepackage{fvextra}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{breaklines,commandchars=\\\{\}}

% Inside a callout, drop the grey code background: the shaded box can't
% break across pages there, so a long listing in an exercise spills out of
% the callout and over the page footer. Unshaded, it breaks like text.
% Deferred: Quarto loads tcolorbox after this file.
\AtBeginDocument{\tcbset{before upper={\renewenvironment{Shaded}{}{}}}}

% Cap figures at 45% of the text height (pandoc's default is the full page),
% so the tall ones, like the flowcharts, sit next to the text that introduces them.
\makeatletter
\renewcommand*\pandocbounded[1]{%
  \sbox\pandoc@box{#1}%
  \Gscale@div\@tempa{0.45\textheight}{\dimexpr\ht\pandoc@box+\dp\pandoc@box\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\pandoc@box}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi
  \ifdim\@tempa\p@<\p@\scalebox{\@tempa}{\usebox\pandoc@box}%
  \else\usebox{\pandoc@box}%
  \fi}
\makeatother
